# Issue #3345 P0: spike script appendix

The P0 spike ran out-of-tree in a gitignored scratch directory so that no mainline
manifest, lockfile, or `node_modules` was modified. The scripts are reproduced
here rather than committed as files, because this repository does not accept new
`.js`/`.mjs` sources.

To reproduce:

```bash
mkdir -p tmp/issue3345/spike && cd tmp/issue3345/spike
npm init -y
npm install ink@7.1.1 ink-scroll-view@0.3.7 ink-scroll-list@0.4.1 ink-virtual-list@0.3.0 --no-save
# write each script below into this directory, then run the commands in its heading
```

Paths beginning `/Users/acoliver/projects/llxprt/branch-7/llxprt-code` are the
repository root at capture time; substitute your own checkout.

## Reproduce the compile spike

From the repository root, generate the temporary configuration by copying and
merging the package configuration rather than creating a small configuration
with `extends`:

```bash
cd packages/cli
node --input-type=module <<'NODE'
import {readFileSync, writeFileSync} from 'node:fs';

const source = readFileSync('tsconfig.json', 'utf8');
const withoutLineComments = source.replace(/^\s*\/\/.*(?:\r?\n|$)/gm, '');
const config = JSON.parse(withoutLineComments);

config.compilerOptions = {
  ...config.compilerOptions,
  paths: {
    ...config.compilerOptions.paths,
    ink: ['../../tmp/issue3345/spike/node_modules/ink/build/index.d.ts'],
  },
  noEmit: true,
  incremental: false,
};
delete config.compilerOptions.tsBuildInfoFile;

writeFileSync('tsconfig.spike3345.json', `${JSON.stringify(config, null, 2)}\n`);
NODE

npx tsc --noEmit -p tsconfig.json > ../../tmp/issue3345/baseline-typecheck.txt 2>&1 || true
npx tsc --noEmit -p tsconfig.spike3345.json > ../../tmp/issue3345/spike-typecheck.txt 2>&1 || true
diff -u ../../tmp/issue3345/baseline-typecheck.txt ../../tmp/issue3345/spike-typecheck.txt
```

The script reads `packages/cli/tsconfig.json`, strips its `//` comment lines,
sets `compilerOptions.paths.ink` to the out-of-tree upstream declaration, sets
`noEmit: true` and `incremental: false`, deletes `tsBuildInfoFile`, and writes
`packages/cli/tsconfig.spike3345.json`. The two `tsc` commands run from
`packages/cli`; `|| true` preserves both diagnostic listings for the diff.

The configuration must be generated by merging. A separate configuration that
uses `extends` to replace `compilerOptions.paths` would replace that object
wholesale and drop the workspace aliases. Delete the temporary
`tsconfig.spike3345.json` after reproduction.

## `static-retention.mjs`: E9 renderer-level `<Static>` accumulation

```js
// Spike (issue #3345): measure renderer-level <Static> accumulation.
// Usage: node static-retention.mjs <absolute path to an ink build dir>
import { EventEmitter } from 'node:events';
import React from 'react';

const inkBuild = process.argv[2];
const iterations = Number(process.argv[3] ?? 300);

const { default: render } = await import(`${inkBuild}/render.js`);
const { default: Box } = await import(`${inkBuild}/components/Box.js`);
const { default: Text } = await import(`${inkBuild}/components/Text.js`);
const { default: Static } = await import(`${inkBuild}/components/Static.js`);

class Stdout extends EventEmitter {
  columns = 80;
  rows = 24;
  isTTY = true;
  bytes = 0;
  write = (chunk) => {
    this.bytes += chunk.length;
  };
}

class Stdin extends EventEmitter {
  isTTY = true;
  setEncoding() {}
  setRawMode() {}
  resume() {}
  pause() {}
  ref() {}
  unref() {}
  read() {
    return null;
  }
}

const stdout = new Stdout();
const line = 'x'.repeat(200);

function App({ items }) {
  return React.createElement(
    Box,
    { flexDirection: 'column' },
    React.createElement(
      Static,
      { items, key: 'static' },
      (item) => React.createElement(Text, { key: item.id }, item.text),
    ),
    React.createElement(Text, null, `live ${items.length}`),
  );
}

const instance = render(React.createElement(App, { items: [] }), {
  stdout,
  stdin: new Stdin(),
  stderr: new Stdout(),
  debug: false,
  exitOnCtrlC: false,
  patchConsole: false,
  // Upstream 7.x only: force interactive. Ignored by the 6.4.8 fork.
  interactive: true,
});

const items = [];
for (let i = 0; i < iterations; i++) {
  items.push({ id: i, text: `${i} ${line}` });
  instance.rerender(React.createElement(App, { items: [...items] }));
}

// Reach the renderer instance through the module-level instance registry so we
// can read the accumulator directly rather than inferring it from heap noise.
const { default: instancesModule } = await import(`${inkBuild}/instances.js`);
const entry = instancesModule.get(stdout);
const ink = entry?.ink ?? entry;

console.log(
  JSON.stringify(
    {
      inkBuild,
      iterations,
      staticBytesEmitted: items.reduce((n, i) => n + i.text.length + 1, 0),
      fullStaticOutputLength: ink?.fullStaticOutput?.length ?? null,
      interactive: ink?.interactive ?? null,
      stdoutBytesWritten: stdout.bytes,
    },
    null,
    2,
  ),
);

instance.unmount();
process.exit(0);
```

## `text-cache-isolated.mjs`: E11: text-cache boundedness, renderer excluded

```js
// Spike (issue #3345): isolate the text-layer caches from renderer/React
// retention by calling the text helpers directly.
// Usage: node --expose-gc text-cache-isolated.mjs <ink build dir> <variant>
//   variant: "upstream" | "fork"
const inkBuild = process.argv[2];
const variant = process.argv[3];
const iterations = Number(process.argv[4] ?? 50_000);

const filler = 'lorem ipsum dolor sit amet consectetur adipiscing elit '.repeat(4);

const heapAfterGc = () => {
  globalThis.gc?.();
  globalThis.gc?.();
  return process.memoryUsage().heapUsed;
};

let call;
if (variant === 'upstream') {
  const { default: wrapText } = await import(`${inkBuild}/wrap-text.js`);
  const { default: measureText } = await import(`${inkBuild}/measure-text.js`);
  call = (text) => {
    measureText(text);
    wrapText(text, 80, 'wrap');
  };
} else {
  const { toStyledCharacters, measureStyledChars } = await import(
    `${inkBuild}/measure-text.js`
  );
  const { wrapStyledChars } = await import(`${inkBuild}/wrap-text.js`);
  call = (text) => {
    const chars = toStyledCharacters(text);
    measureStyledChars(chars);
    wrapStyledChars(chars, 80);
  };
}

// Warm up so one-time module allocations are not counted.
for (let i = 0; i < 100; i++) call(`warmup-${i} ${filler}`);

const baseline = heapAfterGc();
for (let i = 0; i < iterations; i++) call(`${i} ${filler}`);
const after = heapAfterGc();

console.log(
  JSON.stringify(
    {
      inkBuild,
      variant,
      iterations,
      distinctInputBytes: iterations * (filler.length + 7),
      heapGrowthBytes: after - baseline,
    },
    null,
    2,
  ),
);
```
